SELECT Tamu.NamaLengkap, Reservasi.TanggalCheckIn, Reservasi.TanggalCheckOut
FROM Tamu
JOIN Reservasi
ON Tamu.IDTamu = Reservasi.IDTamu;

SELECT 
    Pegawai.NamaLengkap,
    LayananTambahan.NamaLayanan,
    PemesananLayanan.TotalBiaya
FROM PemesananLayanan
JOIN Pegawai ON Pegawai.IDPegawai = PemesananLayanan.IDPegawai
JOIN LayananTambahan ON LayananTambahan.IDLayanan = PemesananLayanan.IDLayanan;

SELECT 
    UPPER(NamaLengkap) AS Nama_Besar,
    YEAR(TanggalCheckIn) AS Tahun_CheckIn
FROM Tamu
JOIN Reservasi ON Tamu.IDTamu = Reservasi.IDTamu;
